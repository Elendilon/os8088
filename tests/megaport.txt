# tests/megaport.txt - MartyPC soak rows PORTED to kern_mega.
#
# Each line registers `mega-<name>` (tests/suite.py's _mega_ports): the SAME
# script as soak row <name>, run with OS88_MEGA=1, which makes
# os88marty.launch() answer tools/os88bochsmarty.py - Bochs on build/mega.img
# with build/megaapps.img in B: (docs/BOCHS.md, "Rows ported from MartyPC").
#
#     name [secs=N] [timeout=N] [+arg ...] [-arg ...] [-flag=]   # why
#
# `secs` is MEASURED on the port (Bochs runs the guest at real time, so a row
# that is guest-heavy costs more than its MartyPC figure); without one it is
# twice the base row's. A row is listed here only once it PASSES on kern_mega
# - and was seen to go red when the thing it checks was broken - or once its
# failure is a kern_mega defect that is fixed in the same change.

sheetbtn secs=60   # Sheet's dialog buttons (Goto/Format) open, close and reuse their slot
uilayer secs=120   # every os88ui verb confirmed off guest state, and confirm-vs-settle

# FILES, DIALOGS AND THE DESK
fpgnotch secs=30   # dsk_xfer asks once a run whether a bar is live; its .cold breakpoints run above 1MB here
deskmodal secs=40   # a widget DELETE with a Disk window in front; scratch system disk off mega.img
deskcalgrid secs=40   # the Desk Calendar grid out of its bss, the clock set through clk_*
msel secs=150   # multi-select and set operations; the listing segment is a selector here
fcpapi secs=30   # OSAPI_FILE_COPY both verbs, every answer read off the volume
fcpcopy secs=90   # Cut/Copy/Paste moves a file and a folder tree, the volume verified on the host
icostore secs=60   # two volumes, one icon-store body; the listing claim is a selector here
desksc secs=150   # desktop shortcuts, every route on one boot, two reboots (a power cycle under Bochs)
imgmount secs=200   # a .IMG mounted by double-click; Unmount Image sends the window to the host (KERN-MEGA-PLAN 14.77)
diskclone secs=90   # Clone Disk end to end, both floppies diffed on the host; the same-drive identity check (KERN-MEGA-PLAN 14.78)
czjoin secs=200   # Uncompress on a part joins the set; every refusal used to hang (KERN-MEGA-PLAN 14.79)
fdlgchoose secs=120   # the Standard File chooser end to end through Note Pad
fdlggrey secs=60   # the chooser's default button greyed with nothing selected; from the root of B: under Bochs, MEDIA overflowing the 1.44MB disk's list
sbardlg secs=60   # the chooser's scroll bar at the list's right edge
facescan144 secs=40   # ty_scan walks to SYSTEM/FONTS - the 1.44MB leg, which is kern_mega's only system disk (facescan's 360KB one has no faces and tests the other rung)
assocopen secs=40   # opening a document draws no pixel of the Disk window; the double-click's second half one X chain under Bochs
assocwake secs=60   # a document launch draws the program's window before reading the document; the glass is fbuf under Bochs
assocfind secs=90   # an unclaimed extension is looked for, the search's disk reads counted
assocflush secs=480   # what the machine learns is written back to ASSOC.DAT, across five boots; the .cold flush breakpoints above 1MB here
