# tests/megaport.txt - MartyPC soak rows PORTED to kern_mega.
#
# Each line registers `mega-<name>` (tests/suite.py's _mega_ports): the SAME
# script as soak row <name>, run with OS88_MEGA=1, which makes
# os88marty.launch() answer tools/os88bochsmarty.py - Bochs on build/mega.img
# with build/megaapps.img in B: (docs/BOCHS.md, "Rows ported from MartyPC").
#
#     name [secs=N] [timeout=N] [+arg ...] [-arg ...] [-flag=]   # why
#
# `secs` is MEASURED on the port (Bochs runs the guest at real time, so a row
# that is guest-heavy costs more than its MartyPC figure); without one it is
# twice the base row's. A row is listed here only once it PASSES on kern_mega
# - and was seen to go red when the thing it checks was broken - or once its
# failure is a kern_mega defect that is fixed in the same change.

sheetbtn secs=60   # Sheet's dialog buttons (Goto/Format) open, close and reuse their slot
uilayer secs=120   # every os88ui verb confirmed off guest state, and confirm-vs-settle

# WINDOW MANAGER AND INPUT
bptrace secs=70   # os88ui driven with breakpoints ARMED (bp_trace) - the shim's bp_exec, pump and stepped double-click
sizesnap secs=20   # the SIZE snap keeps a maximized window at x=0, w=[vid_pw]
btnmove secs=30   # a standard button answers where its window IS after a drag
dragui secs=40   # every control Skies laid out answers after a drag replay
fmarrows secs=50   # Disk-window arrows move a selection; click, Down, click stays inside FM_DBLCLK and opens nothing
zonedmg secs=30   # a zone's damage owes the window over it a redraw (the glass is fbuf on Bochs's VGA)
deskzoom secs=60 +--machine +os8088_xt_vga   # a zoom's restore and an overflowing drag leave a whole repaint - VGA leg only (no Hercules on Bochs)
deskclip secs=90   # a desktop cell is drawn only where it shows (VGA legs; a boot a leg - no snapshot on Bochs)
dmgcull secs=70   # a marked window does not paint under the mover (the Task Manager's PKG_MEGA arm read off its own map)
zoomsave secs=170   # a zoom over another window banks it, the reveal comes off the cache
clipkeep secs=70   # a wholly covered window keeps its raise cache when it arms a clip
uidrdis secs=90   # a drop-down drawn disabled takes no press (two boots: Bochs runs patched code stale; --clobber-dis goes red)
gfxlk secs=70   # nothing draws or banks with the arrow on the glass (a GFXAUDIT=1 kern_mega in a tree of its own)
thm95 secs=240 +--legs +ACDE   # the Windows theme: buttons pixel by pixel, menus, the module's doors (no Hercules leg B; the title bar's 8088 cost skipped)
fsxclip secs=60   # an fsx bracket entered from a click comes back to a whole desktop (--clobber-clip goes red)
altenter secs=60   # Alt+Enter full screen by latch (ArtfulType) and bracket (Tracker) - the bracket entered with a human's hold, which found Tracker unseeded (KERN-MEGA-PLAN 14.77)
